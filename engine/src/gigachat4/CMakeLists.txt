cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(strata_gigachat4 LANGUAGES CXX)
  option(STRATA_BUILD_TESTS "Build the reduced numerical tests" ON)
  option(STRATA_ENABLE_CUDA "Build native CUDA weight projection support" OFF)
  if(STRATA_ENABLE_CUDA)
    enable_language(CUDA)
  endif()
  if(STRATA_BUILD_TESTS)
    enable_testing()
  endif()
endif()

add_library(strata_gigachat4_core STATIC decoder.cpp dsa.cpp gdn.cpp weights.cpp)
target_compile_features(strata_gigachat4_core PUBLIC cxx_std_17)
target_include_directories(strata_gigachat4_core PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/strata-llama/vendor")
target_include_directories(strata_gigachat4_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
if(STRATA_ENABLE_CUDA AND NOT STRATA_HIP_GFX906)
  find_package(CUDAToolkit REQUIRED)
  target_sources(strata_gigachat4_core PRIVATE cuda_backend.cu)
  set_target_properties(strata_gigachat4_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    CUDA_RUNTIME_LIBRARY Static)
  target_compile_definitions(strata_gigachat4_core PUBLIC STRATA_GIGACHAT4_CUDA=1)
  target_link_libraries(strata_gigachat4_core PRIVATE CUDA::cudart_static)
endif()

add_executable(strata-gigachat4 main.cpp)
target_link_libraries(strata-gigachat4 PRIVATE strata_gigachat4_core)

if(STRATA_BUILD_TESTS)
  set(_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/gigachat4")
  foreach(part decoder dsa gdn weights)
    add_executable(gigachat4_${part}_test "${_tests}/${part}_test.cpp")
    target_link_libraries(gigachat4_${part}_test PRIVATE strata_gigachat4_core)
    add_test(NAME gigachat4_${part} COMMAND gigachat4_${part}_test)
  endforeach()
  if(STRATA_ENABLE_CUDA AND NOT STRATA_HIP_GFX906)
    add_executable(gigachat4_cuda_test "${_tests}/cuda_test.cu")
    set_target_properties(gigachat4_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_link_libraries(gigachat4_cuda_test PRIVATE strata_gigachat4_core)
    add_test(NAME gigachat4_cuda COMMAND gigachat4_cuda_test)
    set_tests_properties(gigachat4_cuda PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
